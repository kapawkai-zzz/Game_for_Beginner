bag = {
    "ยา": 3,
    "แอปเปิ้ล": 5,
    "ระเบิด": 1
}

while True:

    print("\n========== INVENTORY ==========")
    print("1. ดูกระเป๋า")
    print("2. ใช้ไอเท็ม")
    print("3. เก็บไอเท็ม")
    print("4. ทิ้งไอเท็ม")
    print("5. ออกเกม")

    # ================= เลือกเมนู =================
    try:
        user = int(input("กรุณาเลือกเมนู : "))

    except ValueError:
        print("กรุณาใส่เป็นตัวเลข")
        continue

    # ================= ดูกระเป๋า =================
    if user == 1:

        print("\n========== กระเป๋า ==========")

        if len(bag) == 0:
            print("กระเป๋าว่าง")

        else:
            for i in bag:
                print(f"{i} มี {bag[i]} ชิ้น")

    # ================= ใช้ไอเท็ม =================
    elif user == 2:

        use = input("กรุณาเลือกไอเท็มที่ต้องการใช้ : ")

        # ---------- ไม่มีไอเท็ม ----------
        if use not in bag:
            print("ไม่มีไอเท็มนี้")
            continue

        # ---------- ของหมด ----------
        elif bag[use] <= 0:
            print("ไอเท็มหมดแล้ว")
            continue

        # ---------- ใช้สำเร็จ ----------
        bag[use] -= 1

        print(f"ใช้ {use} สำเร็จ")
        print(f"เหลือ {bag[use]} ชิ้น")

        # ---------- ถ้าของหมดให้ลบ ----------
        if bag[use] == 0:
            del bag[use]
            print(f"{use} หมดจากกระเป๋าแล้ว")

    # ================= เก็บไอเท็ม =================
    elif user == 3:

        new_item = input("พิมพ์ชื่อไอเท็มที่ต้องการเก็บ : ")

        try:
            amount = int(input("จำนวนที่ต้องการเก็บ : "))

        except ValueError:
            print("กรุณาใส่จำนวนเป็นตัวเลข")
            continue

        # ---------- ถ้าจำนวนติดลบ ----------
        if amount <= 0:
            print("จำนวนต้องมากกว่า 0")
            continue

        # ---------- มีอยู่แล้ว ----------
        if new_item in bag:
            bag[new_item] += amount

        # ---------- ยังไม่มี ----------
        else:
            bag[new_item] = amount

        print(f"ได้รับ {new_item} จำนวน {amount} ชิ้น")

    # ================= ทิ้งไอเท็ม =================
    elif user == 4:

        choose = input("พิมพ์ไอเท็มที่ต้องการทิ้ง : ")

        # ---------- ไม่มีไอเท็ม ----------
        if choose not in bag:
            print("ไม่มีไอเท็มนี้")
            continue

        try:
            amount = int(input("จำนวนที่ต้องการทิ้ง : "))

        except ValueError:
            print("กรุณาใส่จำนวนเป็นตัวเลข")
            continue

        # ---------- จำนวนผิด ----------
        if amount <= 0:
            print("จำนวนต้องมากกว่า 0")
            continue

        # ---------- ของไม่พอ ----------
        if amount > bag[choose]:
            print("จำนวนไอเท็มไม่พอ")
            continue

        # ---------- ทิ้งของ ----------
        bag[choose] -= amount

        print(f"ทิ้ง {choose} จำนวน {amount} ชิ้นแล้ว")

        # ---------- ถ้าหมดให้ลบ ----------
        if bag[choose] == 0:
            del bag[choose]
            print(f"{choose} ถูกลบออกจากกระเป๋าแล้ว")

    # ================= ออกเกม =================
    elif user == 5:

        print("ออกจากเกม...")
        break

    # ================= เมนูผิด =================
    else:
        print("กรุณาเลือกเมนูให้ถูกต้อง")
